--SOLUTION 0001 : A Pretty Simple Handcrank
Select
	A.[Location] As [Location],
	(Case When Sum(F.[Stock_Amount]) Is Null Then 0 Else Sum(F.[Stock_Amount]) End) As [Stock_Amount]
From @Auto_Referenced_Table A
Left Join @Auto_Referenced_Table B
	On (B.[Higher_Level_Location] = A.[Location])
Left Join @Auto_Referenced_Table C
	On (C.[Higher_Level_Location] = B.[Location])
Left Join @Auto_Referenced_Table D
	On (D.[Higher_Level_Location] = C.[Location])
Left Join @Auto_Referenced_Table E
	On (E.[Higher_Level_Location] = D.[Location])
Left Join @Stock_Table F
	On (F.[Location] = A.[Location] Or F.[Location] = B.[Location] Or F.[Location] = C.[Location] Or F.[Location] = D.[Location] Or F.[Location] = E.[Location])
Group By A.[Location]

--SOLUTION 0002 : A More Complex Dynamic Hierarchy, Automatic Level Count Flexor
Declare @Intermediate_Crossjoin_Table Table ([Higher_Level_Location] Varchar(Max), [Location] Varchar(Max), [Stock_Amount] Bigint)
Insert Into @Intermediate_Crossjoin_Table ([Higher_Level_Location], [Location], [Stock_Amount])
Select
	Min(E.[Location]) As [Higher_Level_Location],
	A.[Location] As [Location],
	Sum(D.[Stock_Amount]) As [Stock_Amount]
From @Auto_Referenced_Table A
Cross Join @Auto_Referenced_Table B
Cross Join @Auto_Referenced_Table C
Left Join @Stock_Table D
	On (C.[Higher_Level_Location] = B.[Location] And B.[Higher_Level_Location] = A.[Location] And D.[Location] = A.[Location] Or C.[Higher_Level_Location] = B.[Location] And B.[Higher_Level_Location] = A.[Location] And D.[Location] = B.[Location] Or C.[Higher_Level_Location] = B.[Location] And B.[Higher_Level_Location] = A.[Location] And D.[Location] = C.[Location])
Left Join @Auto_Referenced_Table E
	On (E.[Higher_Level_Location] = A.[Location])
Group By A.[Location]
