Declare @Element_Element_Element_Output Table ([Propal] Varchar(Max), [Order] Varchar(Max), [Invoice] Varchar(Max))
Insert Into @Element_Element_Element_Output ([Propal], [Order], [Invoice])
Select
	C.[REF] As [Propal],
	D.[REF] As [Order],
	E.[REF] As [Invoice]
From @Element_Element A
Inner Join @Element_Element B
	On (A.[Source_Type] = 'propal' And A.[Target_Type] = 'order' And B.[Source_Type] = 'order' And B.[Target_Type] = 'invoice' And B.[FK_Source] = A.[FK_Target])
Inner Join @Propals C
	On (C.[ID] = A.[FK_Source])
Inner Join @Orders D
	On (D.[ID] = A.[FK_Target])
Inner Join @Invoices E
	On (E.[ID] = B.[FK_Target])

Declare @Element_Element_Output Table ([Propal] Varchar(Max), [Order] Varchar(Max), [Invoice] Varchar(Max))
Insert Into @Element_Element_Output ([Propal], [Order], [Invoice])
Select
	B.[REF] As [Propal],
	(Case When C.[REF] Is Not Null Then C.[REF] Else D.[REF] End) As [Order],
	E.[REF] As [Invoice]
From @Element_Element A
Left Join @Propals B
	On (A.[Source_Type] = 'propal' And A.[Target_Type] = 'order' And B.[ID] = A.[FK_Source])
Left Join @Orders C
	On (A.[Source_Type] = 'propal' And A.[Target_Type] = 'order' And C.[ID] = A.[FK_Target])
Left Join @Orders D
	On (A.[Source_Type] = 'order' And A.[Target_Type] = 'invoice' And D.[ID] = A.[FK_Source])
Left Join @Invoices E
	On (A.[Source_Type] = 'order' And A.[Target_Type] = 'invoice' And E.[ID] = A.[FK_Target])
